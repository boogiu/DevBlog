```cpp
HANDLE hFile = CreateFile(
	L".../Data/Line.dat", //경로
	GENERIC_WRITE, // 파일 접근 모드 (읽기 / 쓰기)
	NULL, //공유 방식 : 파일이 개방된 상태에서 다른 프로세스에서 사용하고자 할 때 허가할 것인가  점점... 안중요하다고 하심
	NULL, //보안 속성(기본 보안 상태 유지)
	CREATE_ALWAYS, //생성 방식 (->파일 없으면 생성 , 있으면 덮어쓰기)
	//	OPEN_EXISTING, //생성 방식 (->파일 없으면 읽어오지마!)
	FILE_ATTRIBUTE_NORMAL, // 파일 속성(일반적인 파일 / 읽기 전용  / 숨김 파일 등)
	NULL // 생성될 파일의 속성을 제공할 파일
);
```
